# Shared GPU idle gate for VIctory benchmark windows. Dot-source this file; it only
# defines functions. Used by bin/gpu-quiet and by the investigate-r1 window scripts.
#
# Admission (2026-11-27, user request): the desktop alone keeps the whole GPU at
# 3-6 %, so a run starts when
#   * the last 10 samples average <= 8 % whole-GPU utilization,
#   * no single one of them exceeds 10 %,
#   * no process outside the desktop-background list uses a 3D/Compute/CUDA engine,
#   * no miner process exists.
# Desktop background (compositor, browsers, Edge WebView, Teams, Windows shell) is
# admitted and every such process is written to the log. Anything else only makes
# the gate wait. The gate never stops or signals any process.

function Get-GpuGateDesktop { @(
    'dwm', 'csrss', 'explorer', 'ShellExperienceHost', 'StartMenuExperienceHost',
    'SearchHost', 'SearchApp', 'TextInputHost', 'ApplicationFrameHost', 'LockApp',
    'Microsoft.AAD.BrokerPlugin', 'PhoneExperienceHost', 'Widgets', 'WidgetBoard',
    'chrome', 'msedge', 'msedgewebview2', 'firefox', 'brave', 'opera',
    'ms-teams', 'Teams', 'cloudmusic'
) }

function Get-GpuGateSample {
    $smi = "$env:WINDIR\System32\nvidia-smi.exe"
    $query = '--query-gpu=name,driver_version,utilization.gpu,memory.used,memory.total,power.draw,temperature.gpu,clocks.current.graphics,clocks.current.memory'
    $rows = @(& $smi $query '--format=csv,noheader,nounits')
    if ($LASTEXITCODE -ne 0 -or $rows.Count -ne 1) { throw 'Expected one NVIDIA GPU from nvidia-smi' }
    $fields = $rows[0] -split ',\s*'
    if ($fields.Count -ne 9 -or $fields[2] -notmatch '^\d+$') { throw "Invalid GPU utilization: $rows" }
    $util = [int]$fields[2]
    $eng = @(Get-CimInstance Win32_PerfFormattedData_GPUPerformanceCounters_GPUEngine -OperationTimeoutSec 6)
    $desktop = Get-GpuGateDesktop
    $owners = @(); $other = @(); $tolerated = @(); $tracked = 0
    foreach ($row in $eng) {
        if ($row.Name -notmatch '^pid_(\d+)_.*_engtype_(.+)$') { continue }
        $procId = $Matches[1]; $type = $Matches[2]
        $isTracked = $type -match '^(3d|compute|cuda)'
        if ($isTracked) { $tracked++ }
        if ($null -eq $row.UtilizationPercentage) { throw 'Unreadable GPU Engine counter' }
        if ($row.UtilizationPercentage -le 0) { continue }
        $name = (Get-Process -Id $procId -ErrorAction SilentlyContinue).ProcessName
        if (!$name) { $name = 'unknown' }
        $entry = [pscustomobject]@{ name = $name; pid = $procId; type = $type; pct = [int]$row.UtilizationPercentage }
        if (!$isTracked) { $other += $entry }
        elseif ($name -in $desktop) { $tolerated += $entry }
        else { $owners += $entry }
    }
    if (!$tracked) { throw 'No Windows 3D/Compute engine counters available' }
    $miner = @(Get-Process -Name 'krig-miner', 'alpha-miner' -ErrorAction SilentlyContinue)
    $fmt = { param($list) if ($list.Count) { ($list | ForEach-Object { "$($_.name)(pid=$($_.pid)) $($_.type)=$($_.pct)%" }) -join ', ' } else { 'none' } }
    $line = '{0:u} | {1} | driver={2} | GPU={3}% | VRAM={4}/{5} MiB | power={6} W | temp={7} C | graphics={8} MHz | memory={9} MHz | 3D/Compute={10} | krig-miner={11} | other engines={12} | tolerated background={13}' -f `
        [datetime]::UtcNow, $fields[0], $fields[1], $util, $fields[3], $fields[4], $fields[5], $fields[6], $fields[7], $fields[8],
        (& $fmt $owners), $(if ($miner.Count) { ($miner | ForEach-Object { "$($_.ProcessName)($($_.Id))" }) -join ',' } else { 'none' }),
        (& $fmt $other), (& $fmt $tolerated)
    [pscustomobject]@{ util = $util; owners = $owners; tolerated = $tolerated; miners = $miner.Count; line = $line }
}

# Waits until the admission rule holds. Returns the accepted window, or $null after
# TimeoutSeconds (0 = wait indefinitely). Waiting lines go to the output so the log
# shows why a run was delayed (Write-Host, so they never mix into the returned
# object); at most one per distinct reason every 20 s.
function Wait-GpuIdle {
    param([int]$Samples = 10, [double]$MaxMean = 8, [int]$MaxSingle = 10,
          [double]$MinSpanSeconds = 10, [int]$IntervalMs = 850, [int]$TimeoutSeconds = 0)
    $clock = [Diagnostics.Stopwatch]::StartNew()
    $window = [Collections.Generic.List[object]]::new()
    $busy = 0; $lastReason = ''; $lastLogged = -1e9; $last = $null
    while ($true) {
        $s = Get-GpuGateSample
        $last = $s
        $now = $clock.Elapsed.TotalSeconds
        $reason = $null; $kind = ''
        if ($s.miners) { $reason = 'miner process present'; $kind = 'miner' }
        elseif ($s.owners.Count) { $kind = 'owner'; $reason = 'non-desktop process on 3D/Compute: ' + (($s.owners | ForEach-Object { "$($_.name)(pid=$($_.pid)) $($_.type)=$($_.pct)%" }) -join ', ') }
        elseif ($s.util -gt $MaxSingle) { $kind = 'single'; $reason = "GPU=$($s.util)% > single-sample limit $MaxSingle%" }
        if ($reason) {
            $window.Clear()
        } else {
            $window.Add([pscustomobject]@{ t = $now; s = $s })
            while ($window.Count -gt $Samples) { $window.RemoveAt(0) }
            if ($window.Count -eq $Samples) {
                $vals = @($window | ForEach-Object { $_.s.util })
                $mean = ($vals | Measure-Object -Average).Average
                $span = $now - $window[0].t
                if ($mean -le $MaxMean -and $span -ge $MinSpanSeconds) {
                    $bg = @{}
                    foreach ($w in $window) { foreach ($e in $w.s.tolerated) {
                        $k = "$($e.name)(pid=$($e.pid)) $($e.type)"
                        if (!$bg.ContainsKey($k)) { $bg[$k] = @{ max = 0; n = 0 } }
                        $bg[$k].max = [Math]::Max($bg[$k].max, $e.pct); $bg[$k].n++
                    } }
                    $bgText = if ($bg.Count) { ($bg.Keys | Sort-Object | ForEach-Object { "$_ max=$($bg[$_].max)% in $($bg[$_].n)/$Samples" }) -join ', ' } else { 'none' }
                    return [pscustomobject]@{
                        samples = $vals; mean = $mean; max = ($vals | Measure-Object -Maximum).Maximum
                        span = $span; waited = $now; busy = $busy; background = $bgText; line = $s.line
                        rule = "mean<=$MaxMean% single<=$MaxSingle% over $Samples samples, non-desktop 3D/Compute=none"
                    }
                }
                if ($mean -gt $MaxMean) { $kind = 'mean'; $reason = ('mean of last {0} = {1:F1}% > {2}%' -f $Samples, $mean, $MaxMean) }
            }
        }
        if ($reason) {
            $busy++
            if ($kind -ne $lastReason -or ($now - $lastLogged) -ge 20) {
                Write-Host ("GPU gate waiting ({0:F0}s): {1} | {2}" -f $now, $reason, $s.line)
                $lastReason = $kind; $lastLogged = $now
            }
        }
        if ($TimeoutSeconds -gt 0 -and $now -ge $TimeoutSeconds) {
            Write-Host $last.line
            return $null
        }
        Start-Sleep -Milliseconds $IntervalMs
    }
}

# One-line form for the per-run gates inside window scripts. The prefix is kept
# from the old 3 % gates ("GPU gate samples=").
function Invoke-GpuGate {
    $g = Wait-GpuIdle
    Write-Output ('GPU gate samples={0} mean={1:F1}% max={2}% busy={3} waited_s={4:F1} rule={5} desktop background={6}' -f `
        ($g.samples -join ','), $g.mean, $g.max, $g.busy, $g.waited, $g.rule, $g.background)
}
